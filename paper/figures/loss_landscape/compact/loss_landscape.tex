\newcommand{\lossheight}[2]{%
  1.82
  + 0.075*((#1)^2 + 0.88*(#2)^2)
  - 0.120*(#1)*(#2)
  + 0.135*(#1)
  - 0.055*(#2)
  + 1.95*exp(-(((#1)+2.10)^2)/0.28 - (((#2)-1.70)^2)/0.42)
  + 0.78*exp(-(((#1)-1.76)^2)/0.62 - (((#2)+1.62)^2)/0.52)
  + 1.05*exp(-(((#1)+0.35)^2)/0.36 - (((#2)+2.05)^2)/0.30)
  + 1.34*exp(-(((#1)-0.22)^2)/0.30 - (((#2)-1.18)^2)/0.34)
  + 0.42*exp(-(((#1)-0.90)^2)/1.30 - (((#2)-0.40)^2)/0.90)
  - 1.12*exp(-(((#1)-1.24)^2)/0.52 - (((#2)-1.34)^2)/0.48)

  - 0.70*exp(-(((#1)-1.90)^2)/0.34 - (((#2)+0.18)^2)/0.34)
  + 0.22*exp(-(((#1)-2.15)^2)/0.55 - (((#2)-2.05)^2)/0.48)%
}

\newcommand{\drawdensitytile}[4]{%

  \pgfmathtruncatemacro{\densityshade}{min(88,max(4,round(7 + 265*\densitysample)))}%
  \pgfmathsetmacro{\tilegrow}{0.004}%
  \pgfmathsetmacro{\ua}{(#1)-\tilegrow}%
  \pgfmathsetmacro{\ub}{(#2)+\tilegrow}%
  \pgfmathsetmacro{\va}{(#3)-\tilegrow}%
  \pgfmathsetmacro{\vb}{(#4)+\tilegrow}%

    fill=densityred!\densityshade!white,
    draw=densityred!\densityshade!white,

\newcommand{\drawlosstile}[4]{%
  \pgfmathsetmacro{\lossu}{0.5*((#1)+(#2))}%
  \pgfmathsetmacro{\lossv}{0.5*((#3)+(#4))}%
  \pgfmathsetmacro{\losssample}{\lossheight{\lossu}{\lossv}}%
  \pgfmathtruncatemacro{\lossshade}{min(100,max(0,round(100*(\losssample-0.25)/4.05)))}%
  \pgfmathsetmacro{\surfgrow}{0.002}%
  \pgfmathsetmacro{\ua}{(#1)-\surfgrow}%
  \pgfmathsetmacro{\ub}{(#2)+\surfgrow}%
  \pgfmathsetmacro{\va}{(#3)-\surfgrow}%
  \pgfmathsetmacro{\vb}{(#4)+\surfgrow}%
  \path[
    fill=losslow!\lossshade!losshigh,
    draw=none,
    fill opacity=1
  ]
    \projectpoint{\ua}{\va}{\lossheight{\ua}{\va}} --
    \projectpoint{\ub}{\va}{\lossheight{\ub}{\va}} --
    \projectpoint{\ub}{\vb}{\lossheight{\ub}{\vb}} --
    \projectpoint{\ua}{\vb}{\lossheight{\ua}{\vb}} -- cycle;
}
